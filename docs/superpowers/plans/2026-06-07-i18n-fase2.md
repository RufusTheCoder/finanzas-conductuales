# i18n Fase 2 (pt-BR contenido) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Add a fully usable pt-BR experience for course **content** (the 880 translation rows) plus a language selector (ES/PT) with auto-detect and per-user persistence — without regressing es-MX.

**Architecture:** Two independent tracks. **A (content):** Claude drafts pt-BR for all 880 keys → side-by-side TSV → Rodrigo reviews/edits → import → seed `lang='pt-BR'` to Supabase. **B (selector):** a `switchLang()` helper + `setUserLang()` REST call + an ES/PT toggle in the dashboard nav and auth shell + `navigator.language` auto-detect at boot. Fallback (pt-BR missing → es-MX) already exists from Fase 1b, so everything degrades gracefully. The selector is **deployed after** pt-BR is seeded so "PT" isn't empty.

**Tech Stack:** Vanilla ES modules (no build). Supabase REST. Node (ESM) for data scripts. Verification: node unit tests + `verify_esmx.mjs` (Playwright) + SQL coverage. No test runner in repo.

- [ ] **Step 3: Create `scripts/ptbr_seed.mjs`**

```js
// pt-BR seeding pipeline. Subcommands:
//   node scripts/ptbr_seed.mjs tsv            > scripts/i18n_ptbr_review.tsv
//     (merges buildRows() es-MX with scripts/i18n_ptbr_draft.json → review TSV)
//   node scripts/ptbr_seed.mjs import <tsv>   > scripts/i18n_seed_ptbr.sql
//   node scripts/ptbr_seed.mjs coverage       (compares DB pt-BR keys vs buildRows keys)
import { readFile } from 'node:fs/promises';
import { fileURLToPath } from 'node:url';
import { buildRows } from './extract_translations.mjs';

const TAB = '\t';
const escTsvCell = (s) => String(s ?? '').replace(/\\/g, '\\\\').replace(/\t/g, '\\t').replace(/\r?\n/g, '\\n');
const unescTsvCell = (s) => String(s ?? '').replace(/\\n/g, '\n').replace(/\\t/g, '\t').replace(/\\\\/g, '\\');

export function rowsToTsv(rows) {
  const header = ['key', 'domain', 'es_mx', 'pt_br'].join(TAB);
  const lines = rows.map(r => [r.key, r.domain, escTsvCell(r.es_mx), escTsvCell(r.pt_br)].join(TAB));
  return [header, ...lines].join('\n') + '\n';
}

export function parseTsv(tsv) {
  const lines = tsv.split('\n').filter(l => l.length);
  const [, ...body] = lines; // drop header
  return body.map(line => {
    const [key, domain, es_mx, pt_br] = line.split(TAB);
    return { key, domain, es_mx: unescTsvCell(es_mx), pt_br: unescTsvCell(pt_br) };
  });
}

export const esc = (v) => v === null || v === undefined ? 'NULL' : "'" + String(v).replace(/'/g, "''") + "'";

export function emitSql(rows) {
  const out = [];
  out.push('-- pt-BR seed (generated by ptbr_seed.mjs import)');
  out.push(`-- rows: ${rows.length}`);
  out.push('BEGIN;');
  const CHUNK = 100;
  for (let i = 0; i < rows.length; i += CHUNK) {
    const slice = rows.slice(i, i + CHUNK);
    out.push('INSERT INTO public.translations (key, lang, text, domain) VALUES');
    out.push(slice.map((r, j) =>
      `  (${esc(r.key)}, 'pt-BR', ${esc(r.pt_br)}, ${esc(r.domain)})${j === slice.length - 1 ? '' : ','}`
    ).join('\n'));
    out.push('ON CONFLICT (key, lang) DO NOTHING;');
  }
  out.push('COMMIT;');
  return out.join('\n') + '\n';
}

async function main() {
  const [cmd, arg] = process.argv.slice(2);
  if (cmd === 'tsv') {
    const es = buildRows().filter(r => r.lang === 'es-MX');
    let draft = {};
    try { draft = JSON.parse(await readFile(new URL('./i18n_ptbr_draft.json', import.meta.url), 'utf8')); } catch {}
    const rows = es.map(r => ({ key: r.key, domain: r.domain, es_mx: r.text, pt_br: draft[r.key] ?? '' }));
    process.stdout.write(rowsToTsv(rows));
  } else if (cmd === 'import') {
    const tsv = await readFile(arg, 'utf8');
    const rows = parseTsv(tsv).filter(r => r.pt_br && r.pt_br.trim());
    process.stdout.write(emitSql(rows));
  } else if (cmd === 'coverage') {
    const { SUPABASE_URL, SUPABASE_ANON_KEY } = await import('../public/js/config.js');
    const want = new Set(buildRows().filter(r => r.lang === 'es-MX').map(r => r.key));
    const res = await fetch(`${SUPABASE_URL}/rest/v1/translations?lang=eq.pt-BR&select=key`, {
      headers: { apikey: SUPABASE_ANON_KEY, Authorization: `Bearer ${SUPABASE_ANON_KEY}` },
    });
    const have = new Set((await res.json()).map(r => r.key));
    let missing = 0, extra = 0;
    for (const k of want) if (!have.has(k)) { missing++; console.log('MISSING pt-BR:', k); }
    for (const k of have) if (!want.has(k)) { extra++; console.log('EXTRA pt-BR:', k); }
    console.log(`pt-BR coverage: ${want.size} expected, ${have.size} in DB | missing=${missing} extra=${extra}`);
    process.exit(missing + extra === 0 ? 0 : 1);
  } else {
    console.error('usage: ptbr_seed.mjs tsv | import <tsv> | coverage');
    process.exit(2);
  }
}
const isMain = process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1];
if (isMain) main();
```

```js
function langToggleHtml() {
  const cur = getLang();
  const btn = (code, label) =>
    `<button data-lang="${code}" class="lang-btn" aria-pressed="${cur === code}" style="border:none;background:${cur === code ? 'var(--ink,#1F1F1F)' : 'transparent'};color:${cur === code ? '#fff' : 'var(--ink-4,#888)'};font:600 12px/1 -apple-system,sans-serif;padding:5px 9px;border-radius:6px;cursor:pointer">${label}</button>`;
  return `<div class="lang-toggle" style="display:inline-flex;gap:2px;border:1px solid var(--line,#e5e5e5);border-radius:8px;padding:2px">${btn('es-MX', 'ES')}${btn('pt-BR', 'PT')}</div>`;
}

async function switchLang(lang) {
  if (lang === getLang()) return;
  setLang(lang);
  try { localStorage.setItem('fc_lang', lang); } catch (_) {}
  await loadTranslations(lang);
  render();
  // Don't write another user's lang while impersonating (view_as). state.viewAs is
  // the in-app read-only signal; supabase.js also guards setUserLang with readOnly.
  if (state.user?.email && !state.viewAs) { setUserLang(state.user.email, lang).catch(() => {}); }
}
```

- [ ] **Step 3: Add a delegated click listener** (once, at boot — add right before the final `render();` call near line 2950)

```js
document.addEventListener('click', (e) => {
  const b = e.target.closest('[data-lang]');
```

- [ ] **Step 4: Auto-detect in `bootLang`** — change the boot `bootLang` IIFE (added in Fase 1b) to fall back to browser language:

```js
// OLD
const bootLang = (() => { try { return localStorage.getItem('fc_lang') || 'es-MX'; } catch (_) { return 'es-MX'; } })();
// NEW
const bootLang = (() => {
  try {
    const saved = localStorage.getItem('fc_lang');
    if (saved) return saved;
    return (navigator.language || '').toLowerCase().startsWith('pt') ? 'pt-BR' : 'es-MX';
  } catch (_) { return 'es-MX'; }
})();
```
